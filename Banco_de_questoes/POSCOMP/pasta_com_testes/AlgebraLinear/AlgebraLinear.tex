\def\AMCbeginQuestion#1#2{\par\noindent{\bf (POSCOMP 2023 - Questão 01)}}
\begin{question}{} Utilize o método de Eliminação de Gauss para resolver o sistema a seguir:
\[
\begin{array}{rcl}
-3x+y+z&=&1\\
x-2y+z&=&4\\
-x+y-3z&=&-7
\end{array}
\]
\begin{choices}
\correctchoice{$(0,-1,2)$}
\wrongchoice{$(0,1,0)$}
\wrongchoice{$(2,0,2)$}
\wrongchoice{$(2,1,2)$}
\wrongchoice{$(-1,-1,-1)$}
\end{choices}
\end{question}
}
