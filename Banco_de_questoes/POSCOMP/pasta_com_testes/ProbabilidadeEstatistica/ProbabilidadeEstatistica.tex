\def\AMCbeginQuestion#1#2{\par\noindent{\bf (POSCOMP 2004 - Questão 06)}}
\begin{question}{} Considere o espaço amostral $\Omega=\{\omega_1,\omega_2,\ldots,\omega_n\}$ onde $\omega_i$ ocorre com probabilidade $p_i$ para todo $i\in\{1,2,\ldots,n\}$. Defina o produto escalar
\[
\langle x,y\rangle=p_1x_1y_1+p_2x_2y_2+\cdots+p_nx_ny_n,
\]
para $x=(x_1,x_2,\ldots,x_n)$ e $y=(y_1,y_2,\ldots,y_n)$, pontos quaisquer no $\mathrm{R}^{n}$. Seja $X$ uma variável aleatória com $X(\omega_i)=X_i$. Para $p=(p_1,\ldots,p_n)$, $X=(X_1,\ldots,X_n)$ e $\mathbf{1}=(1,1,\ldots,1)\in\mathrm{R}^{n}$ podemos dizer que
\[
\langle X,\mathbf{1}\rangle,\qquad
\langle X-\langle X,\mathbf{1}\rangle\mathbf{1},X-\langle X,\mathbf{1}\rangle\mathbf{1}\rangle,\qquad
\|X-\langle X,\mathbf{1}\rangle\mathbf{1}\|
\]
são, respectivamente, com respeito à variável $X$ a
\begin{choices}
\correctchoice{média, variância, desvio padrão}
\wrongchoice{variância, média, desvio padrão}
\wrongchoice{média, desvio padrão, variância}
\wrongchoice{desvio padrão, média, variância}
\wrongchoice{desvio padrão, variância, média}
\end{choices}
\end{question}
